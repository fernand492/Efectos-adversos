    // Función que valida la respuesta de cada pregunta
    function checkAnswer(qNumber) {
      var questionDiv = document.querySelector('.question[data-question="' + qNumber + '"]') ||
                        document.getElementById("result" + qNumber).parentElement;
      var correct = document.querySelector('.question:nth-of-type(' + qNumber + ')').getAttribute('data-correct');
      var radios = document.getElementsByName("q" + qNumber);
      var resultDiv = document.getElementById("result" + qNumber);
      var explanationDiv = document.getElementById("explanation" + qNumber);
      var answered = false;
      var selectedValue = "";
      for (var i = 0; i < radios.length; i++) {
        if (radios[i].checked) {
          answered = true;
          selectedValue = radios[i].value;
          break;
        }
      }
      if (!answered) {
        alert("Por favor, selecciona una respuesta en la Pregunta " + qNumber);
        return;
      }
      if (selectedValue === correct) {
        resultDiv.textContent = "¡Correcto!";
        resultDiv.style.color = "green";
      } else {
        resultDiv.textContent = "Incorrecto.";
        resultDiv.style.color = "red";
      }
      explanationDiv.style.display = "block";
    }
